模块一：TI-LightGCN 召回

论文公式 代码实现 说明
式(4-1): 边权重函数 compute_edge_weight() w(u,i) = Σ α_b · exp(-λ·Δt/T)
式(4-3)(4-4): 加权归一化 weighted_propagate() e_u^(l) = Σ w(u,i)/√W_u·√W_i · e_i^(l-1)
式(4-5): 均值聚合 mean_aggregate() ẽ_u = 1/(L+1) Σ e_u^(l)
式(4-6): BPR损失 bpr_loss() -Σ log σ(ẽ_u^T ẽ_i - ẽ_u^T ẽ_j)
式(4-7): 长尾正则 long_tail_regularization() γ · Σ ||ẽ_i - ẽ_i^sem||²
# ti_lightgcn.py - TI-LightGCN 核心实现 import torch import torch.nn as nn import torch.nn.functional as F import numpy as np from torch_geometric.nn import MessagePassing # 行为类型语义权重（表4-1） BEHAVIOR_WEIGHTS = { 'buy': 3.0, 'collect': 1.5, 'cart': 1.2, 'click': 1.0, 'show': 0.1 } # 时间衰减参数 LAMBDA = 0.1 # 时间衰减率 T_NORMALIZE = 86400 # 时间归一化常数（秒/天） class TILightGCN(MessagePassing): """TI-LightGCN: 时间感知行为加权图协同过滤""" def __init__(self, num_users, num_items, embedding_dim=64, num_layers=3): super(TILightGCN, self).__init__(aggr='add') self.num_users = num_users self.num_items = num_items self.embedding_dim = embedding_dim self.num_layers = num_layers # 初始化嵌入矩阵（Xavier初始化） self.user_embedding = nn.Embedding(num_users, embedding_dim) self.item_embedding = nn.Embedding(num_items, embedding_dim) nn.init.xavier_uniform_(self.user_embedding.weight) nn.init.xavier_uniform_(self.item_embedding.weight) def compute_edge_weight(self, interactions): """ 式(4-1): w(u,i) = Σ α_b · exp(-λ · (T_now - t) / T) Args: interactions: List of (user_id, item_id, behavior_type, timestamp) Returns: edge_weights: Dict[(user_id, item_id)] -> weight """ edge_weights = {} T_now = interactions[-1][3] # 当前时间 for user_id, item_id, behavior, timestamp in interactions: alpha_b = BEHAVIOR_WEIGHTS[behavior] delta_t = T_now - timestamp weight = alpha_b * np.exp(-LAMBDA * delta_t / T_NORMALIZE) key = (user_id, item_id) edge_weights[key] = edge_weights.get(key, 0) + weight return edge_weights def forward(self, edge_index, edge_weights, user_degree, item_degree): """ 式(4-3)(4-4): 加权归一化传播 e_u^(l) = Σ w(u,i)/√W_u · √W_i · e_i^(l-1) """ all_embeddings = [torch.cat([self.user_embedding.weight, self.item_embedding.weight], dim=0)] for l in range(self.num_layers): # 计算加权归一化系数 norm_weights = [] for idx, (u, i) in enumerate(edge_index.t().tolist()): w_ui = edge_weights[idx] W_u = user_degree[u] W_i = item_degree[i] norm = w_ui / (np.sqrt(W_u) * np.sqrt(W_i)) norm_weights.append(norm) norm_weights = torch.tensor(norm_weights, dtype=torch.float32) # 消息传递 embeddings = all_embeddings[-1] new_embeddings = self.propagate(edge_index, x=embeddings, edge_weight=norm_weights) all_embeddings.append(new_embeddings) # 式(4-5): 均值聚合 final_embedding = torch.stack(all_embeddings, dim=1).mean(dim=1) # 分离用户和物品嵌入 user_emb = final_embedding[:self.num_users] item_emb = final_embedding[self.num_users:] return user_emb, item_emb def message(self, x_j, edge_weight): """加权消息传递""" return edge_weight.view(-1, 1) * x_j def bpr_loss(user_emb, pos_item_emb, neg_item_emb): """ 式(4-6): BPR损失 L_BPR = -Σ log σ(ẽ_u^T ẽ_i - ẽ_u^T ẽ_j) """ pos_scores = (user_emb * pos_item_emb).sum(dim=1) neg_scores = (user_emb * neg_item_emb).sum(dim=1) loss = -torch.log(torch.sigmoid(pos_scores - neg_scores)).mean() return loss def long_tail_regularization(item_emb, item_sem_emb, tail_mask, gamma=0.01): """ 式(4-7): 长尾度感知正则化 L_total = L_BPR + γ · Σ ||ẽ_i - ẽ_i^sem||² Args: item_emb: 物品图嵌入 item_sem_emb: 物品语义嵌入（Sentence-BERT） tail_mask: 长尾商品掩码（交互次数<10） gamma: 正则化强度 """ tail_item_emb = item_emb[tail_mask] tail_sem_emb = item_sem_emb[tail_mask] reg_loss = F.mse_loss(tail_item_emb, tail_sem_emb.detach()) return gamma * reg_loss # 训练主函数 def train_ti_lightgcn(interactions, num_users, num_items, epochs=50, lr=0.001): """TI-LightGCN训练流程""" device = torch.device('cuda' if torch.cuda.is_available() else 'cpu') # 1. 构建加权图 edge_weights = compute_edge_weight(interactions) edge_index, weights = build_graph(edge_weights, num_users, num_items) edge_index = edge_index.to(device) weights = weights.to(device) # 2. 计算加权度 user_degree = compute_weighted_degree(edge_index, weights, 'user') item_degree = compute_weighted_degree(edge_index, weights, 'item') # 3. 初始化模型 model = TILightGCN(num_users, num_items).to(device) optimizer = torch.optim.Adam(model.parameters(), lr=lr, weight_decay=1e-4) # 4. 训练循环 best_recall = 0 patience_counter = 0 for epoch in range(epochs): model.train() optimizer.zero_grad() # 前向传播 user_emb, item_emb = model(edge_index, weights, user_degree, item_degree) # 采样BPR三元组 pos_pairs, neg_items = sample_bpr_triplets(interactions, num_items) pos_item_emb = item_emb[pos_pairs[:, 1]] neg_item_emb = item_emb[neg_items] user_emb_batch = user_emb[pos_pairs[:, 0]] # 计算损失 loss_bpr = bpr_loss(user_emb_batch, pos_item_emb, neg_item_emb) loss_tail = long_tail_regularization(item_emb, item_sem_emb, tail_mask) loss = loss_bpr + loss_tail # 反向传播 loss.backward() optimizer.step() # 验证早停 if (epoch + 1) % 5 == 0: recall = evaluate_recall(model, val_data) if recall > best_recall: best_recall = recall patience_counter = 0 torch.save(model.state_dict(), 'best_model.pt') else: patience_counter += 1 if patience_counter >= 5: print(f"Early stopping at epoch {epoch+1}") break return model

模块二：Temporal-DIN 排序

公式对照表

论文公式 代码实现 说明
式(4-8): 时间衰减注意力 temporal_attention() α_k = softmax(a(e_ik, e_ic)) · exp(-γ·Δt/T)
式(4-9): 归一化 softmax() ã_k = α_k / Σ α_j
式(4-10): 短期兴趣 short_term_interest() h_st = Σ ã_k · e_ik
式(4-11): 门控 gating_fusion() g = sigmoid(W_g · [h_st; h_lt; e_ic] + b_g)
式(4-12): 兴趣融合 fuse_interests() h_user = g ⊙ h_st + (1-g) ⊙ h_lt
式(4-13): CTR预测 mlp_predict() ŷ = MLP([h_user; e_ic; e_u^static; e_i^stat])
# temporal_din.py - Temporal-DIN 核心实现 import torch import torch.nn as nn import torch.nn.functional as F # 时间衰减参数 GAMMA = 0.3 # 时间衰减率 T_NORMALIZE = 86400 # 秒/天 T_SHORT = 7 * T_NORMALIZE # 短期兴趣窗口：7天 class TemporalDIN(nn.Module): """Temporal-DIN: 时间衰减兴趣网络""" def __init__(self, num_items, embedding_dim=64, hidden_dim=256): super(TemporalDIN, self).__init__() self.item_embedding = nn.Embedding(num_items, embedding_dim) self.user_static_emb = nn.Embedding(num_users, embedding_dim) self.item_stat_emb = nn.Embedding(num_items, embedding_dim) # 激活单元（DIN注意力） self.activation_unit = nn.Sequential( nn.Linear(embedding_dim * 4, hidden_dim), nn.ReLU(), nn.Linear(hidden_dim, 1) ) # 门控融合 self.gate = nn.Sequential( nn.Linear(embedding_dim * 3, hidden_dim), nn.Sigmoid() ) # MLP预测层 self.mlp = nn.Sequential( nn.Linear(embedding_dim * 4, hidden_dim), nn.ReLU(), nn.Linear(hidden_dim, hidden_dim // 2), nn.ReLU(), nn.Linear(hidden_dim // 2, 1), nn.Sigmoid() ) def temporal_attention(self, hist_items, hist_times, candidate_item, T_now): """ 式(4-8): 时间衰减注意力 α_k = softmax(a(e_ik, e_ic)) · exp(-γ · (T_now - t_k) / T) """ # 获取嵌入 hist_emb = self.item_embedding(hist_items) # [batch, seq_len, dim] cand_emb = self.item_embedding(candidate_item) # [batch, dim] # DIN注意力分数 cand_expanded = cand_emb.unsqueeze(1).expand_as(hist_emb) diff = hist_emb - cand_expanded prod = hist_emb * cand_expanded attention_input = torch.cat([hist_emb, cand_expanded, diff, prod], dim=-1) din_scores = self.activation_unit(attention_input).squeeze(-1) # [batch, seq_len] # 时间衰减因子 delta_t = T_now - hist_times # [batch, seq_len] time_decay = torch.exp(-GAMMA * delta_t / T_NORMALIZE) # 结合 alpha = torch.softmax(din_scores, dim=-1) * time_decay # 式(4-9): 归一化 alpha_normalized = alpha / (alpha.sum(dim=-1, keepdim=True) + 1e-8) return alpha_normalized def forward(self, user_id, hist_items, hist_times, candidate_item, user_memory, T_now, user_static=None, item_stat=None): """ 完整前向传播 """ # 式(4-10): 短期兴趣 alpha = self.temporal_attention(hist_items, hist_times, candidate_item, T_now) hist_emb = self.item_embedding(hist_items) h_st = (alpha.unsqueeze(-1) * hist_emb).sum(dim=1) # [batch, dim] # 长期兴趣（来自User Memory） h_lt = self.memory_projection(user_memory) # [batch, dim] # 式(4-11)(4-12): 门控融合 cand_emb = self.item_embedding(candidate_item) gate_input = torch.cat([h_st, h_lt, cand_emb], dim=-1) g = self.gate(gate_input) h_user = g * h_st + (1 - g) * h_lt # 式(4-13): CTR预测 user_static_emb = self.user_static_emb(user_id) item_stat_emb = self.item_stat_emb(candidate_item) mlp_input = torch.cat([h_user, cand_emb, user_static_emb, item_stat_emb], dim=-1) prediction = self.mlp(mlp_input) return prediction.squeeze(-1) # 蒸馏损失 def distillation_loss(y_pred, y_true, y_teacher, alpha=0.7, beta=0.3): """ 式(4-14)(4-15)(4-16): 联合损失函数 L_total = α · L_beh + β · L_kd """ # 行为损失（二元交叉熵） L_beh = F.binary_cross_entropy(y_pred, y_true) # 蒸馏损失（MSE） L_kd = F.mse_loss(y_pred, y_teacher.detach()) return alpha * L_beh + beta * L_kd

# llm_distillation.py - LLM偏好蒸馏实现 import json from openai import OpenAI from sentence_transformers import SentenceTransformer # 初始化 client = OpenAI(api_key="your-api-key") encoder = SentenceTransformer('paraphrase-multilingual-MiniLM-L12-v2') def generate_user_memory(user_behaviors, user_profile): """ User Memory生成（式4.3.2） Args: user_behaviors: List of (item_name, behavior_type, timestamp) user_profile: Dict with industry, role, etc. Returns: memory_vector: 256-dim numpy array memory_json: Structured preference summary """ # 步骤一：行为历史结构化 behavior_text = format_behaviors(user_behaviors) # 步骤二：LLM推理生成结构化摘要 prompt = f""" 你是一个工业采购专家。请根据以下用户行为历史，生成结构化的偏好摘要。 用户信息： - 行业：{user_profile.get('industry', '未知')} - 注册时间：{user_profile.get('register_time', '未知')} 近30天行为记录： {behavior_text} 请输出JSON格式： {{ "industry_domain": "所属行业领域", "preferred_brands": ["品牌1", "品牌2"], "price_sensitivity": "价格敏感度描述", "tech_keywords": ["技术关键词1", "技术关键词2"], "purchase_intent": "当前采购意图", "long_term_preference": "长期偏好总结" }} """ response = client.chat.completions.create( model="qwen-7b-chat", messages=[{"role": "user", "content": prompt}], temperature=0.3 ) memory_json = json.loads(response.choices[0].message.content) # 步骤三：向量化 memory_text = json.dumps(memory_json, ensure_ascii=False) memory_vector = encoder.encode(memory_text) return memory_vector, memory_json def generate_teacher_labels(user_memory, candidate_items): """ 生成教师信号（教师标签） Returns: relevance_scores: Dict[item_id] -> score in [0, 1] """ prompt = f""" 基于以下用户偏好摘要，请对每个候选商品评估相关性评分（0-1之间）。 用户偏好： {user_memory} 候选商品： {format_items(candidate_items)} 请输出JSON格式：{{item_id: score}} """ response = client.chat.completions.create( model="qwen-7b-chat", messages=[{"role": "user", "content": prompt}], temperature=0.1 ) return json.loads(response.choices[0].message.content) # 蒸馏训练主函数 def train_with_distillation(model, train_data, user_memories, epochs=10, alpha=0.7, beta=0.3): """ 式(4-14): 联合损失训练 """ optimizer = torch.optim.Adam(model.parameters(), lr=0.001) for epoch in range(epochs): for batch in train_data: user_id, hist_items, hist_times, candidates, labels = batch # 学生模型预测 predictions = model(user_id, hist_items, hist_times, candidates, user_memories[user_id], T_now=1700000000) # 获取教师标签（离线预生成） teacher_labels = get_teacher_labels(user_id, candidates) # 联合损失 loss = distillation_loss(predictions, labels.float(), teacher_labels.float(), alpha, beta) optimizer.zero_grad() loss.backward() optimizer.step()
